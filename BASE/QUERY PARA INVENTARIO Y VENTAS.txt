WITH compra_info AS (
    SELECT
        UPPER(TRIM(VEHI_SERIE)) AS VEHI_SERIE,
        MAX(TRIM(MODE_TIPO)) AS MODE_TIPO
    FROM AUTOS.VT_VLIBRODECOMPRASCHEV
    WHERE VEHI_SERIE IS NOT NULL
    GROUP BY UPPER(TRIM(VEHI_SERIE))
),

ventas_por_tipo AS (
    SELECT 
        CASE
            WHEN UPPER(TRIM(vt.DESCMODELO)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(vt.DESCMODELO)), '^[^ ]+')
        END
        || '_' || NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN') AS MODELO,
        vt.EMPR_EMPRESAID AS EMPRESA,
        TO_CHAR(TO_DATE(vt.FAAU_FECHA, 'YYYYMMDD'), 'YYYY-MM') AS MES_ANIO,
        SUM(CASE 
                WHEN vt.FAAU_STATUS = 'AC' THEN 1
                WHEN vt.FAAU_STATUS = 'CA' THEN -1
                ELSE 0
            END) AS VENTAS_NETAS
    FROM AUTOS.VT_VLIBRODEVENTASCONCANUTIL_2 vt
    LEFT JOIN compra_info ci
        ON ci.VEHI_SERIE = UPPER(TRIM(vt.VEHI_SERIE))
    WHERE vt.FAAU_FECHA IS NOT NULL
      AND vt.FAAU_STATUS IN ('AC', 'CA')
      AND TO_DATE(vt.FAAU_FECHA, 'YYYYMMDD') 
            BETWEEN ADD_MONTHS(TRUNC(SYSDATE), -11) AND TRUNC(SYSDATE)
      AND vt.FORM_GRUPODEVENTADIR IN ('MENUDEO')
    GROUP BY 
        vt.EMPR_EMPRESAID,
        CASE
            WHEN UPPER(TRIM(vt.DESCMODELO)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(vt.DESCMODELO)), '^[^ ]+')
        END,
        NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN'),
        TO_CHAR(TO_DATE(vt.FAAU_FECHA, 'YYYYMMDD'), 'YYYY-MM')
),

ventas_pivot AS (
    SELECT
        EMPRESA,
        MODELO,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -11), 'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_FINAL,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -10), 'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_2,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -9),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_3,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -8),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_4,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -7),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_5,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -6),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_6,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -5),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_7,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -4),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_8,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -3),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_9,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -2),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_10,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -1),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_11,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(TRUNC(SYSDATE), 'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS VENTA_MES_ACTUAL,
        SUM(VENTAS_NETAS) AS VTAS_12_MESES
    FROM ventas_por_tipo
    GROUP BY EMPRESA, MODELO
),

ventas_color_base AS (
    SELECT 
        vt.EMPR_EMPRESAID AS EMPRESA,
        CASE
            WHEN UPPER(TRIM(vt.DESCMODELO)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(vt.DESCMODELO)), '^[^ ]+')
        END
        || '_' || NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN') AS MODELO,
        NVL(NULLIF(TRIM(vt.COLOR), ''), 'SIN COLOR') AS COLOR,
        TO_CHAR(TO_DATE(vt.FAAU_FECHA, 'YYYYMMDD'), 'YYYY-MM') AS MES_ANIO,
        SUM(CASE 
                WHEN vt.FAAU_STATUS = 'AC' THEN 1
                WHEN vt.FAAU_STATUS = 'CA' THEN -1
                ELSE 0
            END) AS VENTAS_NETAS
    FROM AUTOS.VT_VLIBRODEVENTASCONCANUTIL_2 vt
    LEFT JOIN compra_info ci
        ON ci.VEHI_SERIE = UPPER(TRIM(vt.VEHI_SERIE))
    WHERE vt.FAAU_FECHA IS NOT NULL
      AND vt.FAAU_STATUS IN ('AC', 'CA')
      AND TO_DATE(vt.FAAU_FECHA, 'YYYYMMDD') 
            BETWEEN ADD_MONTHS(TRUNC(SYSDATE), -11) AND TRUNC(SYSDATE)
      AND vt.FORM_GRUPODEVENTADIR IN ('MENUDEO')
    GROUP BY 
        vt.EMPR_EMPRESAID,
        CASE
            WHEN UPPER(TRIM(vt.DESCMODELO)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(vt.DESCMODELO)), '^[^ ]+')
        END,
        NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN'),
        NVL(NULLIF(TRIM(vt.COLOR), ''), 'SIN COLOR'),
        TO_CHAR(TO_DATE(vt.FAAU_FECHA, 'YYYYMMDD'), 'YYYY-MM')
),

ventas_color_pivot AS (
    SELECT
        EMPRESA,
        MODELO,
        COLOR,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -11), 'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_FINAL,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -10), 'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_2,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -9),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_3,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -8),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_4,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -7),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_5,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -6),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_6,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -5),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_7,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -4),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_8,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -3),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_9,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -2),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_10,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(ADD_MONTHS(TRUNC(SYSDATE), -1),  'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS MES_11,
        SUM(CASE WHEN MES_ANIO = TO_CHAR(TRUNC(SYSDATE), 'YYYY-MM') THEN VENTAS_NETAS ELSE 0 END) AS VENTA_MES_ACTUAL,
        SUM(VENTAS_NETAS) AS VTAS_12_MESES
    FROM ventas_color_base
    GROUP BY EMPRESA, MODELO, COLOR
),

ventas_desc AS (
    SELECT
        vt.EMPR_EMPRESAID AS EMPRESA,
        CASE
            WHEN UPPER(TRIM(vt.DESCMODELO)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(vt.DESCMODELO)), '^[^ ]+')
        END
        || '_' || NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN') AS MODELO,
        MAX(vt.MODE_DESCRIPCION) AS MODE_DESCRIPCION
    FROM AUTOS.VT_VLIBRODEVENTASCONCANUTIL_2 vt
    LEFT JOIN compra_info ci
        ON ci.VEHI_SERIE = UPPER(TRIM(vt.VEHI_SERIE))
    WHERE vt.FAAU_FECHA IS NOT NULL
      AND vt.FAAU_STATUS IN ('AC','CA')
      AND TO_DATE(vt.FAAU_FECHA, 'YYYYMMDD') 
            BETWEEN ADD_MONTHS(TRUNC(SYSDATE), -11) AND TRUNC(SYSDATE)
      AND vt.FORM_GRUPODEVENTADIR IN ('MENUDEO')
    GROUP BY 
        vt.EMPR_EMPRESAID,
        CASE
            WHEN UPPER(TRIM(vt.DESCMODELO)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(vt.DESCMODELO)), '^[^ ]+')
        END,
        NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN')
),

inv_filtrado AS (
    SELECT
        inv.EMPR_EMPRESAID AS EMPRESA,
        CASE
            WHEN UPPER(TRIM(inv.MODE_DESCRIPCION)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(inv.MODE_DESCRIPCION)), '^[^ ]+')
        END
        || '_' || NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN') AS MODELO_KEY,
        CASE
            WHEN UPPER(TRIM(inv.MODE_DESCRIPCION)) LIKE 'CAPTIVA PHEV%' THEN 'CAPTIVA_PHEV'
            ELSE REGEXP_SUBSTR(UPPER(TRIM(inv.MODE_DESCRIPCION)), '^[^ ]+')
        END
        || '_' || NVL(NULLIF(TRIM(ci.MODE_TIPO), ''), 'SIN') AS MODELO_DESC,
        NVL(NULLIF(TRIM(inv.COLEXT), ''), 'SIN COLOR') AS COLOR,
        inv.MODE_CLAVE,
        inv.VEHI_SERIE
    FROM AUTOS.VT_VINVENTARIOAUTOS inv
    LEFT JOIN compra_info ci
        ON ci.VEHI_SERIE = UPPER(TRIM(inv.VEHI_SERIE))
    WHERE inv.VEHI_STATUS = 'AC'
      AND inv.VEHI_CLAS_CLAVE <> 'US'
),

inv_total_modelo AS (
    SELECT
        EMPRESA,
        MODELO_KEY,
        MAX(MODELO_DESC) AS MODE_DESCRIPCION_VENTAS,
        COUNT(DISTINCT VEHI_SERIE) AS TOTAL_INVENTARIO
    FROM inv_filtrado
    GROUP BY EMPRESA, MODELO_KEY
),

inv_por_color AS (
    SELECT
        EMPRESA,
        MODELO_KEY,
        MAX(MODELO_DESC) AS MODE_DESCRIPCION_VENTAS,
        COLOR,
        COUNT(DISTINCT VEHI_SERIE) AS INVENTARIO_POR_COLOR
    FROM inv_filtrado
    GROUP BY EMPRESA, MODELO_KEY, COLOR
),

colores_modelo AS (
    SELECT
        EMPRESA,
        MODELO_KEY,
        MAX(MODE_DESCRIPCION_VENTAS) AS MODE_DESCRIPCION_VENTAS,
        COLOR
    FROM inv_por_color
    GROUP BY EMPRESA, MODELO_KEY, COLOR

    UNION

    SELECT
        EMPRESA,
        MODELO AS MODELO_KEY,
        MAX(MODELO) AS MODE_DESCRIPCION_VENTAS,
        COLOR
    FROM ventas_color_pivot
    GROUP BY EMPRESA, MODELO, COLOR
),

emp_modelo_sin_inv AS (
    SELECT vp.EMPRESA, vp.MODELO
    FROM ventas_pivot vp
    WHERE NVL(vp.VTAS_12_MESES, 0) > 0
      AND NOT EXISTS (
          SELECT 1
          FROM inv_total_modelo itm
          WHERE itm.EMPRESA = vp.EMPRESA
            AND itm.MODELO_KEY = vp.MODELO
      )
)

SELECT
    x.EMPRESA,
    x.MODE_DESCRIPCION_VENTAS,
    x.COLOR,
    x.TOTAL_INVENTARIO,
    x.VTAS_12_MESES,
    x.PROM_VTA_3M,
    x.COBERTURA_3M,
    x.PROM_VTA_6M,
    x.COBERTURA_6M,
    x.VENTA_MES_ACTUAL,
    x.MES_11,
    x.MES_10,
    x.MES_9,
    x.MES_8,
    x.MES_7,
    x.MES_6,
    x.MES_5,
    x.MES_4,
    x.MES_3,
    x.MES_2,
    x.MES_FINAL,
    x.INVENTARIO_POR_COLOR,
    x.ORDEN_COLOR,
    x.MODELO_KEY,
    x.MODE_DESCRIPCION
FROM (
    SELECT
        itm.EMPRESA,
        itm.MODE_DESCRIPCION_VENTAS,
        CAST(NULL AS VARCHAR2(100)) AS COLOR,
        itm.TOTAL_INVENTARIO,
        NVL(vp.VTAS_12_MESES, 0) AS VTAS_12_MESES,
        ROUND((NVL(vp.MES_9,0) + NVL(vp.MES_10,0) + NVL(vp.MES_11,0)) / 3, 2) AS PROM_VTA_3M,
        CASE 
          WHEN (NVL(vp.MES_9,0) + NVL(vp.MES_10,0) + NVL(vp.MES_11,0)) = 0 THEN 0
          ELSE ROUND(itm.TOTAL_INVENTARIO / ROUND((NVL(vp.MES_9,0) + NVL(vp.MES_10,0) + NVL(vp.MES_11,0)) / 3, 2), 2)
        END AS COBERTURA_3M,
        ROUND((NVL(vp.MES_6,0)+NVL(vp.MES_7,0)+NVL(vp.MES_8,0)+NVL(vp.MES_9,0)+NVL(vp.MES_10,0)+NVL(vp.MES_11,0)) / 6, 2) AS PROM_VTA_6M,
        CASE 
          WHEN (NVL(vp.MES_6,0)+NVL(vp.MES_7,0)+NVL(vp.MES_8,0)+NVL(vp.MES_9,0)+NVL(vp.MES_10,0)+NVL(vp.MES_11,0)) = 0 THEN 0
          ELSE ROUND(itm.TOTAL_INVENTARIO / ROUND((NVL(vp.MES_6,0)+NVL(vp.MES_7,0)+NVL(vp.MES_8,0)+NVL(vp.MES_9,0)+NVL(vp.MES_10,0)+NVL(vp.MES_11,0)) / 6, 2), 2)
        END AS COBERTURA_6M,

        CAST(NULL AS NUMBER) AS VENTA_MES_ACTUAL,
        CAST(NULL AS NUMBER) AS MES_11,
        CAST(NULL AS NUMBER) AS MES_10,
        CAST(NULL AS NUMBER) AS MES_9,
        CAST(NULL AS NUMBER) AS MES_8,
        CAST(NULL AS NUMBER) AS MES_7,
        CAST(NULL AS NUMBER) AS MES_6,
        CAST(NULL AS NUMBER) AS MES_5,
        CAST(NULL AS NUMBER) AS MES_4,
        CAST(NULL AS NUMBER) AS MES_3,
        CAST(NULL AS NUMBER) AS MES_2,
        CAST(NULL AS NUMBER) AS MES_FINAL,

        CAST(NULL AS NUMBER) AS INVENTARIO_POR_COLOR,
        0 AS ORDEN_COLOR,
        itm.MODELO_KEY,
        vd.MODE_DESCRIPCION
    FROM inv_total_modelo itm
    LEFT JOIN ventas_pivot vp
      ON vp.EMPRESA = itm.EMPRESA
     AND vp.MODELO = itm.MODELO_KEY
    LEFT JOIN ventas_desc vd
      ON vd.EMPRESA = itm.EMPRESA
     AND vd.MODELO = itm.MODELO_KEY

    UNION ALL

    SELECT
        cm.EMPRESA,
        cm.MODE_DESCRIPCION_VENTAS,
        cm.COLOR,
        CAST(NULL AS NUMBER) AS TOTAL_INVENTARIO,
        NVL(vcp.VTAS_12_MESES, 0) AS VTAS_12_MESES,
        CAST(NULL AS NUMBER) AS PROM_VTA_3M,
        CAST(NULL AS NUMBER) AS COBERTURA_3M,
        CAST(NULL AS NUMBER) AS PROM_VTA_6M,
        CAST(NULL AS NUMBER) AS COBERTURA_6M,
        NVL(vcp.VENTA_MES_ACTUAL, 0) AS VENTA_MES_ACTUAL,
        NVL(vcp.MES_11, 0) AS MES_11,
        NVL(vcp.MES_10, 0) AS MES_10,
        NVL(vcp.MES_9, 0) AS MES_9,
        NVL(vcp.MES_8, 0) AS MES_8,
        NVL(vcp.MES_7, 0) AS MES_7,
        NVL(vcp.MES_6, 0) AS MES_6,
        NVL(vcp.MES_5, 0) AS MES_5,
        NVL(vcp.MES_4, 0) AS MES_4,
        NVL(vcp.MES_3, 0) AS MES_3,
        NVL(vcp.MES_2, 0) AS MES_2,
        NVL(vcp.MES_FINAL, 0) AS MES_FINAL,
        NVL(ipc.INVENTARIO_POR_COLOR, 0) AS INVENTARIO_POR_COLOR,
        1 AS ORDEN_COLOR,
        cm.MODELO_KEY,
        CAST(NULL AS VARCHAR2(200)) AS MODE_DESCRIPCION
    FROM colores_modelo cm
    LEFT JOIN inv_por_color ipc
      ON ipc.EMPRESA = cm.EMPRESA
     AND ipc.MODELO_KEY = cm.MODELO_KEY
     AND ipc.COLOR = cm.COLOR
    LEFT JOIN ventas_color_pivot vcp
      ON vcp.EMPRESA = cm.EMPRESA
     AND vcp.MODELO = cm.MODELO_KEY
     AND vcp.COLOR = cm.COLOR

    UNION ALL

    SELECT
        emi.EMPRESA,
        vp.MODELO AS MODE_DESCRIPCION_VENTAS,
        CAST(NULL AS VARCHAR2(100)) AS COLOR,
        0 AS TOTAL_INVENTARIO,
        NVL(vp.VTAS_12_MESES, 0) AS VTAS_12_MESES,
        ROUND((NVL(vp.MES_9,0) + NVL(vp.MES_10,0) + NVL(vp.MES_11,0)) / 3, 2) AS PROM_VTA_3M,
        0 AS COBERTURA_3M,
        ROUND((NVL(vp.MES_6,0)+NVL(vp.MES_7,0)+NVL(vp.MES_8,0)+NVL(vp.MES_9,0)+NVL(vp.MES_10,0)+NVL(vp.MES_11,0)) / 6, 2) AS PROM_VTA_6M,
        0 AS COBERTURA_6M,

        CAST(NULL AS NUMBER) AS VENTA_MES_ACTUAL,
        CAST(NULL AS NUMBER) AS MES_11,
        CAST(NULL AS NUMBER) AS MES_10,
        CAST(NULL AS NUMBER) AS MES_9,
        CAST(NULL AS NUMBER) AS MES_8,
        CAST(NULL AS NUMBER) AS MES_7,
        CAST(NULL AS NUMBER) AS MES_6,
        CAST(NULL AS NUMBER) AS MES_5,
        CAST(NULL AS NUMBER) AS MES_4,
        CAST(NULL AS NUMBER) AS MES_3,
        CAST(NULL AS NUMBER) AS MES_2,
        CAST(NULL AS NUMBER) AS MES_FINAL,

        CAST(NULL AS NUMBER) AS INVENTARIO_POR_COLOR,
        0 AS ORDEN_COLOR,
        vp.MODELO AS MODELO_KEY,
        vd.MODE_DESCRIPCION
    FROM emp_modelo_sin_inv emi
    JOIN ventas_pivot vp
      ON vp.EMPRESA = emi.EMPRESA
     AND vp.MODELO = emi.MODELO
    LEFT JOIN ventas_desc vd
      ON vd.EMPRESA = vp.EMPRESA
     AND vd.MODELO = vp.MODELO
) x
ORDER BY
    x.EMPRESA,
    x.MODE_DESCRIPCION_VENTAS,
    x.ORDEN_COLOR,
    x.COLOR
